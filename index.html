Quick Install

Download and install into your local binary PATH via curl or PowerShell.

# Download latest release for your OS and install to /usr/local/bin
curl -sSL https://ado-board-cli.0101.party/install.sh | bash
# Download latest zip release from GitHub Releases and unpack
Invoke-WebRequest -Uri "https://github.com/jagadeesh-k-ti/ado-board-cli-releases/releases/latest/download/ado-board-cli_windows_amd64.zip" -OutFile "ado-board-cli.zip"
Expand-Archive -Path "ado-board-cli.zip" -DestinationPath "$env:LOCALAPPDATA\Programs\ado-board-cli"
